## Install

```sh
npx stylesheet-ui add sign-up-form otp-input
```

## Usage

Sign-up is usually two steps: create the account, then verify the emailed code with [OtpInput](/components/auth/otp-input/).

```tsx
import { useState } from "react";
import { SignUpForm } from "@/components/ui/sign-up-form";
import { OtpInput } from "@/components/ui/otp-input";

export default function SignUpScreen() {
  const [pending, setPending] = useState(false);
  const [code, setCode] = useState("");
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState<string>();

  const onSubmit = async ({ emailAddress, password }) => {
    setLoading(true);
    try {
      await createAccount(emailAddress, password); // your auth provider
      await sendVerificationCode();                // your auth provider
      setPending(true);
    } catch (e) {
      setError("Couldn't sign up.");
    } finally {
      setLoading(false);
    }
  };

  const verify = async (value: string) => {
    await confirmVerificationCode(value); // your auth provider
  };

  if (pending) return <OtpInput value={code} onChange={setCode} onComplete={verify} error={error} autoFocus />;
  return <SignUpForm onSubmit={onSubmit} loading={loading} error={error} />;
}
```

## Fields

By default the form collects email + password. Two opt-in props add more:

- **`nameFields`** — adds First/Last name inputs (side by side). When set, `onSubmit` receives `firstName` and `lastName` alongside `emailAddress` and `password`.
- **`requireConfirm`** — adds a confirm-password field and blocks submit until it matches.

```tsx
<SignUpForm
  nameFields
  requireConfirm
  onSubmit={({ emailAddress, password, firstName, lastName }) => {
    // create the account with whatever fields you collected
  }}
/>
```

Need other fields (username, phone, terms checkbox)? The form is yours — add inputs and include them in the `onSubmit` payload.
