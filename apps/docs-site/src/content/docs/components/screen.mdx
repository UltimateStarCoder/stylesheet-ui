---
title: Screen
description: SafeAreaView + ScrollView + theme background in one component.
---

## Install

```sh
npx stylesheet-ui add screen
```

## Usage

```tsx
import { Screen } from "@/components/ui/screen";

<Screen padding={16}>
  <YourContent />
</Screen>

<Screen scroll={false}>
  <FixedHeightLayout />
</Screen>
```
